مرجع متدهای IRubix

فهرست ۲۰ متد عمومی کلاس IRubix به‌همراه امضا، توضیح و نمونه کد.

request()get_me()get_chat()get_file()send_text()send_message()send_poll()send_quiz()forward_message()edit_message_text()delete_message()ban_chat_member()unban_chat_member()send_file()command()message()callback()start()stop()run()Event

request()

(self, method: str, input: dict | None = None)

فراخوانی مستقیم هر متد Bot API به‌صورت async.

result = await bot.request("getMe")

get_me()

(self)

دریافت اطلاعات ربات با متد getMe.

me = await bot.get_me()

get_chat()

(self, chat_id: str)

دریافت اطلاعات گفتگو.

chat = await bot.get_chat("CHAT_ID")

get_file()

(self, file_id: str)

دریافت اطلاعات فایل با شناسه فایل.

file = await bot.get_file("FILE_ID")

send_text()

(self, chat_id: str, text: str, **kwargs)

ارسال پیام متنی. گزینه‌های مجاز: metadata، chat_keypad، inline_keypad، reply_to_message_id، disable_notification و chat_keypad_type.

await bot.send_text("CHAT_ID", "سلام")

send_message()

(self, chat_id: str, text: str, **kwargs)

نام جایگزین send_text با همان گزینه‌ها.

await bot.send_message("CHAT_ID", "سلام")

send_poll()

(self, chat_id: str, question: str, options: list[str])

ارسال نظرسنجی؛ حداقل دو گزینه لازم است.

await bot.send_poll("CHAT_ID", "رنگ مورد علاقه؟", ["آبی", "سبز"])

send_quiz()

(self, *args, **kwargs)

در نسخه فعلی پیاده‌سازی نشده و NotImplementedError می‌دهد.

# فعلاً قابل استفاده نیست

forward_message()

(self, from_chat_id: str, message_id: str, to_chat_id: str, disable_notification: bool = False)

فوروارد پیام از یک گفتگو به گفتگوی دیگر.

await bot.forward_message("FROM", "MSG_ID", "TO")

edit_message_text()

(self, chat_id: str, message_id: str, text: str)

ویرایش متن پیام.

await bot.edit_message_text("CHAT_ID", "MSG_ID", "متن جدید")

delete_message()

(self, chat_id: str, message_id: str)

حذف پیام.

await bot.delete_message("CHAT_ID", "MSG_ID")

ban_chat_member()

(self, chat_id: str, user_id: str)

مسدود کردن کاربر در گروه یا کانال.

await bot.ban_chat_member("CHAT_ID", "USER_ID")

unban_chat_member()

(self, chat_id: str, user_id: str)

رفع مسدودیت کاربر.

await bot.unban_chat_member("CHAT_ID", "USER_ID")

send_file()

(self, chat_id: str, path: str | Path | bytes, text: str | None = None, file_type: str = "File", **kwargs)

بارگذاری فایل و ارسال آن. path مسیر فایل یا داده باینری است؛ file_name و گزینه‌های پاسخ/کیبورد نیز قابل ارسال‌اند.

await bot.send_file("CHAT_ID", "photo.jpg", text="تصویر")

command()

(self, names)

ثبت یک یا چند دستور با decorator؛ نام دستور بدون اسلش هم پذیرفته می‌شود.

@bot.command(["start", "help"])

message()

(self, flt=None)

ثبت handler پیام. flt در صورت callable بودن به‌عنوان فیلتر رویداد استفاده می‌شود.

@bot.message()

callback()

(self)

ثبت handler کلیک دکمه‌های inline.

@bot.callback()

start()

(self, poll_interval: float = 1.0, limit: int = 100)

شروع دریافت آپدیت‌ها به‌صورت async.

await bot.start(poll_interval=1.0, limit=100)

stop()

(self)

درخواست توقف حلقه polling.

bot.stop()

run()

(self, poll_interval: float = 1.0, limit: int = 100)

اجرای ربات؛ بیرون event loop خودش اجرا می‌کند و داخل event loop باید مقدار برگشتی را await کنید.

bot.run()

شیء Event

هر handler دو آرگومان می‌گیرد: bot و event.

ویژگیتوضیح
event.botنمونه ربات
event.rawداده خام آپدیت
event.updateهمان داده خام آپدیت
event.typeنوع رویداد
event.update_typeنوع آپدیت
event.messageاطلاعات پیام به شکل دیکشنری
event.chat_idشناسه گفتگو
event.message_idشناسه پیام
event.textمتن پیام یا رشته خالی
event.sender_idشناسه فرستنده
event.button_idشناسه یا داده callback
event.reply(text, **kwargs)پاسخ به پیام فعلی و در صورت وجود، ریپلای به آن

شروع سریع

پس از نصب، توکن ربات خود را وارد کنید و handler دستور شروع را بسازید:

from irubix import IRubix

bot = IRubix("YOUR_BOT_TOKEN")

@bot.command("start")
async def start(bot, event):
    await event.reply("سلام! به ربات من خوش آمدید.")

bot.run()
